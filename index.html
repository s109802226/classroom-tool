function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Points");
  var result = {};
  if (sheet) {
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0]) result[data[i][0]] = data[i][1];
    }
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var logSheet = ss.getSheetByName("Daily_Logs");
    if (!logSheet) {
      logSheet = ss.insertSheet("Daily_Logs");
      logSheet.appendRow(["日期", "出席人數", "缺席人數", "學生明細與點數", "白板公告摘要"]);
    }
    logSheet.appendRow([data.date, data.presentCount, data.absentCount, JSON.stringify(data.studentDetails), data.wbSummary || ""]);
    
    var pointsSheet = ss.getSheetByName("Points");
    if (!pointsSheet) {
      pointsSheet = ss.insertSheet("Points");
      pointsSheet.appendRow(["學生姓名", "歷史累計點數"]);
    }
    if (data.pointsData) {
      var pData = pointsSheet.getDataRange().getValues();
      var nameRowMap = {};
      for (var i = 1; i < pData.length; i++) {
        if (pData[i][0]) nameRowMap[pData[i][0]] = i + 1;
      }
      for (var name in data.pointsData) {
        var addPoints = Number(data.pointsData[name]) || 0;
        if (nameRowMap[name]) {
          var curr = Number(pointsSheet.getRange(nameRowMap[name], 2).getValue()) || 0;
          pointsSheet.getRange(nameRowMap[name], 2).setValue(curr + addPoints);
        } else {
          pointsSheet.appendRow([name, addPoints]);
        }
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
